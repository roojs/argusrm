### 2. New Linux application subclass

```vala
/*

namespace Argus.Agent.Linux
{
	internal class Application : Argus.Agent.Application
	{
		public override int command_line(GLib.ApplicationCommandLine command_line)
		{
			var args = command_line.get_arguments();
			foreach (var arg in args) {
				if (arg == "--help" || arg == "-h") {
					return base.command_line(command_line);
				}
			}
			if (Posix.geteuid() == 0) {
				return base.command_line(command_line);
			}
			var display = GLib.Environment.get_variable("DISPLAY");
			var wayland = GLib.Environment.get_variable("WAYLAND_DISPLAY");
			if ((display == null || display == "") && (wayland == null || wayland == "")) {
				command_line.printerr("Run argus-agent as root.\n");
				return 1;
			}
			var pkexec = GLib.Environment.find_program_in_path("pkexec");
			if (pkexec == null) {
				command_line.printerr("Could not request administrator access: pkexec was not found.\n");
				return 1;
			}
			var executable = "";
			try {
				executable = GLib.FileUtils.read_link("/proc/self/exe");
			} catch (GLib.FileError e) {
				command_line.printerr("Could not resolve argus-agent: %s\n", e.message);
				return 1;
			}
			try {
				new GLib.Subprocess.newv({pkexec, executable}, GLib.SubprocessFlags.NONE);
			} catch (GLib.Error e) {
				command_line.printerr("Could not request administrator access: %s\n", e.message);
				return 1;
			}
			return 0;
		}
	}
}
```

## `agent/linux/org.roojs.ArgusAgent.register.policy.in`

### 3. New Polkit action for the installed executable

**Where:** add the complete new file.

**Depends on:** configured install path from Meson.

#### Add — authorize active desktop administrators

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE policyconfig PUBLIC
 "-//freedesktop//DTD PolicyKit Policy Configuration 1.0//EN"
 "http://www.freedesktop.org/standards/PolicyKit/1/policyconfig.dtd">
<policyconfig>
  <vendor>RooJS</vendor>
  <action id="org.roojs.ArgusAgent.register">
    <description>Set up the Argus agent</description>
    <message>Authentication is required to set up the Argus agent</message>
    <defaults>
      <allow_any>no</allow_any>
      <allow_inactive>no</allow_inactive>
      <allow_active>auth_admin</allow_active>
    </defaults>
    <annotate key="org.freedesktop.policykit.exec.path">@AGENT_EXECUTABLE@</annotate>
    <annotate key="org.freedesktop.policykit.exec.allow_gui">true</annotate>
  </action>
</policyconfig>
```

## `agent/meson.build`

### 4. Linux sources and Polkit action

**Where:** replace the complete current Linux block.

**Depends on:** new template and policy files above.

#### Remove

```meson
if host_machine.system() == 'linux'
	deps += dependency('gio-unix-2.0')
	sources += files(
		'OSQuery/Initialize.vala',
		'OSQuery/Query.vala',
		'linux/Command.vala',
		'Listen.vala',
	)
	vala_args += ['-D', 'LINUX', '--pkg', 'gio-unix-2.0']
endif
```

#### Replace with

```meson
if host_machine.system() == 'linux'
	deps += dependency('gio-unix-2.0')
	agent_executable = get_option('bindir') / 'argus-agent'
	if not agent_executable.startswith('/')
		agent_executable = get_option('prefix') / agent_executable
	endif
	register_policy = configure_file(
		input: 'linux/org.roojs.ArgusAgent.register.policy.in',
		output: 'org.roojs.ArgusAgent.register.policy',
		configuration: {
			'AGENT_EXECUTABLE': agent_executable,
		},
	)
	sources += files(
		'OSQuery/Initialize.vala',
		'OSQuery/Query.vala',
		'linux/Application.vala',
		'linux/Command.vala',
		'Listen.vala',
	)
	vala_args += [
		'-D', 'LINUX', '--pkg', 'gio-unix-2.0', '--pkg', 'posix',
	]
	install_data(
		register_policy,
		install_dir: get_option('datadir') / 'polkit-1' / 'actions',
	)
endif
```
